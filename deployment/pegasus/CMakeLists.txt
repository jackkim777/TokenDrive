cmake_minimum_required(VERSION 3.10)
project(tokendrive_pegasus LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(USE_CUDA "Enable the ONNX Runtime CUDA provider and GPU preprocessing" ON)
option(BUILD_VEHICLE_RUNTIME "Build inference_v3 with DriveWorks camera support" ON)
option(USE_NVMEDIA_STUBS "Use the archived local NvMedia header shims" ON)

# Supply the installed ONNX Runtime API include directory and shared library.
# The vehicle snapshot uses the ONNX Runtime 1.10 API (GetInputName/GetOutputName).
set(ORT_INCLUDE_DIR "" CACHE PATH "Directory containing onnxruntime_cxx_api.h")
set(ORT_LIBRARY "" CACHE FILEPATH "Path to libonnxruntime.so")
if(NOT EXISTS "${ORT_INCLUDE_DIR}/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "Set ORT_INCLUDE_DIR to the directory containing onnxruntime_cxx_api.h")
endif()
if(NOT EXISTS "${ORT_LIBRARY}" OR IS_DIRECTORY "${ORT_LIBRARY}")
    message(FATAL_ERROR "Set ORT_LIBRARY to the ONNX Runtime shared library file")
endif()
get_filename_component(ORT_LIBRARY_DIR "${ORT_LIBRARY}" DIRECTORY)

find_package(Threads REQUIRED)
if(USE_CUDA)
    find_package(CUDA REQUIRED)
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 14)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()

function(link_ort target)
    target_include_directories(${target} PRIVATE "${ORT_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE "${ORT_LIBRARY}" Threads::Threads)
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${ORT_LIBRARY_DIR}")
    if(USE_CUDA)
        target_compile_definitions(${target} PRIVATE USE_CUDA)
        target_include_directories(${target} PRIVATE ${CUDA_INCLUDE_DIRS})
        target_link_libraries(${target} PRIVATE ${CUDA_LIBRARIES})
    endif()
endfunction()

add_executable(benchmark_onnx benchmark_onnx.cpp)
link_ort(benchmark_onnx)

if(BUILD_VEHICLE_RUNTIME)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "inference_v3 requires Linux SocketCAN and the DRIVE SDK")
    endif()
    set(DW_ROOT "/usr/local/driveworks-2.2/targets/aarch64-Linux" CACHE PATH "DriveWorks SDK root")
    set(GLFW_INCLUDE_DIR "/usr/local/driveworks-2.2/samples/3rdparty/src/glfw/3.2-screen/include"
        CACHE PATH "Directory containing GLFW/glfw3.h")
    if(NOT EXISTS "${DW_ROOT}/include/dw/core/Context.h")
        message(FATAL_ERROR "Set DW_ROOT to the installed DriveWorks SDK target root")
    endif()
    if(NOT EXISTS "${GLFW_INCLUDE_DIR}/GLFW/glfw3.h")
        message(FATAL_ERROR "Set GLFW_INCLUDE_DIR to the DriveWorks GLFW include directory")
    endif()

    find_library(DW_LIBRARY driveworks HINTS "${DW_ROOT}/lib")
    find_library(DW_VIZ_LIBRARY driveworks_visualization HINTS "${DW_ROOT}/lib")
    find_library(GLFW_LIBRARY glfw HINTS "${DW_ROOT}/lib")
    find_library(XCURSOR_LIBRARY Xcursor)
    find_library(XRANDR_LIBRARY Xrandr)
    find_library(XINERAMA_LIBRARY Xinerama)
    foreach(dependency DW_LIBRARY DW_VIZ_LIBRARY GLFW_LIBRARY XCURSOR_LIBRARY XRANDR_LIBRARY XINERAMA_LIBRARY)
        if(NOT ${dependency})
            message(FATAL_ERROR "Missing ${dependency}; install the target SDK/display libraries or set its path")
        endif()
    endforeach()

    if(USE_CUDA)
        add_executable(inference_v3 inference_v3.cpp preprocess_cuda.cu)
    else()
        add_executable(inference_v3 inference_v3.cpp)
    endif()
    link_ort(inference_v3)
    target_include_directories(inference_v3 PRIVATE "${DW_ROOT}/include" "${GLFW_INCLUDE_DIR}")
    if(USE_NVMEDIA_STUBS)
        target_include_directories(inference_v3 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/nvmedia_stubs")
    endif()
    target_link_libraries(inference_v3 PRIVATE
        "${DW_LIBRARY}" "${DW_VIZ_LIBRARY}" "${GLFW_LIBRARY}"
        EGL GLESv2 X11 "${XCURSOR_LIBRARY}" "${XRANDR_LIBRARY}" "${XINERAMA_LIBRARY}")
    set_target_properties(inference_v3 PROPERTIES BUILD_RPATH "${ORT_LIBRARY_DIR};${DW_ROOT}/lib")
endif()
